// CONFIGURÁ TUS CREDENCIALES DE SUPABASE ACÁ const SUPABASE_URL = 'TU_SUPABASE_URL'; const SUPABASE_ANON_KEY = 'TU_SUPABASE_ANON_KEY'; const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); let products = []; let cart = []; let accounts = []; let expenses = []; let sales = []; document.addEventListener('DOMContentLoaded', async () => { document.getElementById('today').textContent = new Date().toLocaleDateString(); setupTabs(); setupEvents(); await loadAllData(); }); function setupTabs() { document.querySelectorAll('.tab').forEach(btn => { btn.addEventListener('click', () => { document.querySelectorAll('.tab').forEach(t => t.classList.remove('active')); document.querySelectorAll('main > section').forEach(s => s.classList.add('hidden')); btn.classList.add('active'); document.getElementById(`tab-${btn.dataset.tab}`).classList.remove('hidden'); }); }); } function setupEvents() { document.getElementById('add-product').onclick = addProduct; document.getElementById('add-expense').onclick = addExpense; document.getElementById('add-account').onclick = addAccount; document.getElementById('clear-cart').onclick = () => { cart = []; renderCart(); }; document.getElementById('charge').onclick = processSale; document.getElementById('pay').onchange = (e) => { const box = document.getElementById('account-box'); if (e.target.value === 'cuenta_corriente') box.classList.remove('hidden'); else box.classList.add('hidden'); }; document.getElementById('search-vender').oninput = (e) => renderProducts(e.target.value); document.getElementById('clear-search-vender').onclick = () => { document.getElementById('search-vender').value = ''; renderProducts(); }; document.getElementById('search-stock').oninput = (e) => renderStock(e.target.value); document.getElementById('clear-search-stock').onclick = () => { document.getElementById('search-stock').value = ''; renderStock(); }; } async function loadAllData() { try { const [pRes, aRes, eRes, sRes] = await Promise.all([ supabaseClient.from('products').select('*').order('name'), supabaseClient.from('accounts').select('*').order('name'), supabaseClient.from('expenses').select('*').order('created_at', { ascending: false }), supabaseClient.from('sales').select('*').order('created_at', { ascending: false }).limit(20) ]); products = pRes.data || []; accounts = aRes.data || []; expenses = eRes.data || []; sales = sRes.data || []; renderProducts(); renderStock(); renderAccounts(); renderExpenses(); renderSalesHistory(); renderSummary(); updateAccountSelect(); } catch (err) { console.error('Error cargando datos:', err); } } // --- PRODUCTOS Y VENTA --- function renderProducts(filter = '') { const grid = document.getElementById('product-grid'); grid.innerHTML = ''; const filtered = products.filter(p => p.name.toLowerCase().includes(filter.toLowerCase())); filtered.forEach(p => { const tile = document.createElement('div'); tile.className = 'product-tile'; tile.innerHTML = `
${p.name}
$${p.price} ${p.type === 'weight' ? '/kg' : ''}
`; tile.onclick = () => addToCart(p); grid.appendChild(tile); }); } function addToCart(product) { let qty = 1; if (product.type === 'weight') { const input = prompt(`Ingrese los Kg para ${product.name}:`, '1'); if (!input) return; qty = parseFloat(input); if (isNaN(qty) || qty <= 0) return; } const existing = cart.find(item => item.id === product.id); if (existing) { existing.qty += qty; } else { cart.push({ ...product, qty }); } renderCart(); } function renderCart() { const container = document.getElementById('cart'); container.innerHTML = ''; let total = 0; cart.forEach((item, index) => { const subtotal = item.price * item.qty; total += subtotal; const row = document.createElement('div'); row.style.cssText = 'display:flex; justify-content:space-between; margin-bottom:6px; font-size:0.9rem; align-items:center;'; row.innerHTML = ` ${item.name} (${item.qty} ${item.type === 'weight' ? 'kg' : 'un'}) $${subtotal.toFixed(2)} `; container.appendChild(row); }); document.getElementById('cart-total').textContent = `$${total.toFixed(2)}`; } window.removeFromCart = function(index) { cart.splice(index, 1); renderCart(); }; async function processSale() { if (cart.length === 0) return alert('El carrito está vacío'); const payMethod = document.getElementById('pay').value; const accountId = document.getElementById('account-select').value; if (payMethod === 'cuenta_corriente' && !accountId) { return alert('Seleccioná una cuenta corriente'); } const total = cart.reduce((acc, item) => acc + (item.price * item.qty), 0); const costTotal = cart.reduce((acc, item) => acc + ((item.cost || 0) * item.qty), 0); // Guardar venta const { data: saleData, error } = await supabaseClient.from('sales').insert([{ total, cost: costTotal, payment_method: payMethod, account_id: payMethod === 'cuenta_corriente' ? accountId : null }]).select(); if (error) { alert('Error al registrar venta: ' + error.message); return; } // Si es cuenta corriente, actualizar deuda if (payMethod === 'cuenta_corriente') { const accObj = accounts.find(a => a.id == accountId); if (accObj) { await supabaseClient.from('accounts').update({ debt: (accObj.debt || 0) + total }).eq('id', accountId); } } cart = []; renderCart(); await loadAllData(); alert('¡Venta registrada con éxito!'); } // --- STOCK --- async function addProduct() { const name = document.getElementById('p-name').value.trim(); const price = parseFloat(document.getElementById('p-price').value); const cost = parseFloat(document.getElementById('p-cost').value) || 0; const stock = parseFloat(document.getElementById('p-stock').value) || 0; const type = document.getElementById('p-type').value; if (!name || isNaN(price)) return alert('Completá el nombre y el precio'); const { error } = await supabaseClient.from('products').insert([{ name, price, cost, stock, type }]); if (error) return alert('Error: ' + error.message); document.getElementById('p-name').value = ''; document.getElementById('p-price').value = ''; document.getElementById('p-cost').value = ''; document.getElementById('p-stock').value = '10'; await loadAllData(); } function renderStock(filter = '') { const tbody = document.getElementById('stock-body'); tbody.innerHTML = ''; const filtered = products.filter(p => p.name.toLowerCase().includes(filter.toLowerCase())); filtered.forEach(p => { const tr = document.createElement('tr'); tr.innerHTML = ` ${p.name} $${p.price} $${p.cost || 0} ${p.stock} ${p.type === 'weight' ? 'kg' : 'un'} `; tbody.appendChild(tr); }); } window.deleteProduct = async function(id) { if (!confirm('¿Borrar producto?')) return; await supabaseClient.from('products').delete().eq('id', id); await loadAllData(); }; // --- GASTOS --- async function addExpense() { const description = document.getElementById('e-desc').value.trim(); const amount = parseFloat(document.getElementById('e-amount').value); if (!description || isNaN(amount)) return alert('Completá la descripción y el monto'); const { error } = await supabaseClient.from('expenses').insert([{ description, amount }]); if (error) return alert('Error: ' + error.message); document.getElementById('e-desc').value = ''; document.getElementById('e-amount').value = ''; await loadAllData(); } function renderExpenses() { const tbody = document.getElementById('expense-body'); tbody.innerHTML = ''; expenses.forEach(e => { const tr = document.createElement('tr'); tr.innerHTML = ` ${new Date(e.created_at).toLocaleDateString()} ${e.description} $${e.amount} `; tbody.appendChild(tr); }); } window.deleteExpense = async function(id) { await supabaseClient.from('expenses').delete().eq('id', id); await loadAllData(); }; // --- CUENTAS CORRIENTES --- async function addAccount() { const name = document.getElementById('cc-name').value.trim(); if (!name) return alert('Ingresá un nombre'); const { error } = await supabaseClient.from('accounts').insert([{ name, debt: 0 }]); if (error) return alert('Error: ' + error.message); document.getElementById('cc-name').value = ''; await loadAllData(); } function renderAccounts() { const tbody = document.getElementById('cc-body'); tbody.innerHTML = ''; accounts.forEach(a => { const tr = document.createElement('tr'); tr.innerHTML = ` ${a.name} $${a.debt || 0} `; tbody.appendChild(tr); }); } window.payDebt = async function(id, currentDebt) { const pago = prompt(`Deuda actual: $${currentDebt}. ¿Cuánto paga?`, currentDebt); if (!pago) return; const val = parseFloat(pago); if (isNaN(val) || val <= 0) return; const newDebt = Math.max(0, currentDebt - val); await supabaseClient.from('accounts').update({ debt: newDebt }).eq('id', id); await loadAllData(); }; window.deleteAccount = async function(id) { if (!confirm('¿Eliminar cuenta?')) return; await supabaseClient.from('accounts').delete().eq('id', id); await loadAllData(); }; function updateAccountSelect() { const select = document.getElementById('account-select'); select.innerHTML = ''; accounts.forEach(a => { const opt = document.createElement('option'); opt.value = a.id; opt.textContent = `${a.name} (Deuda: $${a.debt || 0})`; select.appendChild(opt); }); } // --- RESUMEN Y HISTORIAL --- function renderSalesHistory() { const tbody = document.getElementById('sales-body'); tbody.innerHTML = ''; sales.forEach(s => { const tr = document.createElement('tr'); tr.innerHTML = ` ${new Date(s.created_at).toLocaleString()} ${s.payment_method} $${s.total} `; tbody.appendChild(tr); }); } function renderSummary() { const totalSales = sales.reduce((acc, s) => acc + s.total, 0); const totalCost = sales.reduce((acc, s) => acc + (s.cost || 0), 0); const totalProfit = totalSales - totalCost; const totalExpenses = expenses.reduce((acc, e) => acc + e.amount, 0); const net = totalProfit - totalExpenses; document.getElementById('s-sales').textContent = `$${totalSales.toFixed(2)}`; document.getElementById('s-count').textContent = sales.length; document.getElementById('s-profit').textContent = `$${totalProfit.toFixed(2)}`; document.getElementById('s-exp').textContent = `$${totalExpenses.toFixed(2)}`; document.getElementById('s-net').textContent = `$${net.toFixed(2)}`; }
